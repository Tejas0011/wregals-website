function heatColor(val: number, max = 75) {
 const ratio = val / max;
 if (ratio < 0.15) return 'rgba(59, 130, 246, 0.1)';
 if (ratio < 0.35) return 'rgba(59, 130, 246, 0.3)';
 if (ratio < 0.55) return 'rgba(59, 130, 246, 0.5)';
 if (ratio < 0.75) return 'rgba(59, 130, 246, 0.7)';
 return 'rgba(59, 130, 246, 1)';
}
